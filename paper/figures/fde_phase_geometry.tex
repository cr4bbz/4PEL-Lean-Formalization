% Threshold c = 0.6. The three affine paths illustrate positive-first,
% negative-first, and simultaneous N -> B transitions.
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  x=8.5cm,
  y=8.5cm,
  >=Latex,
  line cap=round,
  line join=round,
  font=\small
]
  \def\c{0.6}

  % Four threshold cells.
  \fill[black!5] (0,0) rectangle (\c,\c);
  \fill[green!10] (\c,0) rectangle (1,\c);
  \fill[red!10] (0,\c) rectangle (\c,1);
  \fill[blue!6!red!8] (\c,\c) rectangle (1,1);

  % Unit square, axes, and threshold walls.
  \draw[thick] (0,0) rectangle (1,1);
  \draw[->,thick] (0,0) -- (1.04,0);
  \draw[->,thick] (0,0) -- (0,1.04);
  \draw[dashed,very thick] (\c,0) -- (\c,1);
  \draw[dashed,very thick] (0,\c) -- (1,\c);

  % Axis and wall labels.
  \node[below,font=\bfseries] at (0.50,-0.055)
    {Positive support $(P^{+})$};
  \node[rotate=90,font=\bfseries] at (-0.065,0.50)
    {Negative support $(P^{-})$};
  \node[rotate=90,fill=white,inner sep=1pt,text=black!70]
    at (0.615,0.18) {$P^{+}=c=0.6$};
  \node[fill=white,inner sep=1pt,text=black!70]
    at (0.18,0.615) {$P^{-}=c=0.6$};

  % Cell labels, kept away from the three trajectories.
  \node[align=center,text=black!65] at (0.40,0.14)
    {\textbf{$\Nval$}\\[-1mm]\scriptsize neither / gap};
  \node[align=center,text=green!45!black] at (0.82,0.20)
    {\textbf{$\Tval$}\\[-1mm]\scriptsize true only};
  \node[align=center,text=red!65!black] at (0.18,0.84)
    {\textbf{$\Fval$}\\[-1mm]\scriptsize false only};
  \node[align=center,text=blue!45!red!65!black] at (0.84,0.90)
    {\textbf{$\Bval$}\\[-1mm]\scriptsize both / glut};

  % Positive-first path: (0.2,0.1) -> (0.9,0.7).
  % t_+ = 4/7, t_- = 5/6.
  \coordinate (xone) at (0.20,0.10);
  \coordinate (yone) at (0.90,0.70);
  \coordinate (pplusone) at (0.60,0.443);
  \coordinate (pminusone) at (0.783,0.60);
  \draw[blue!80!black,very thick,-{Latex[length=2.6mm]}]
    (xone) -- (yone);
  \filldraw[blue!80!black] (xone) circle (2.2pt);
  \filldraw[blue!80!black] (yone) circle (2.2pt);
  \filldraw[draw=blue!80!black,fill=white,very thick]
    (pplusone) circle (2.5pt);
  \filldraw[draw=blue!80!black,fill=white,very thick]
    (pminusone) circle (2.5pt);
  \node[blue!80!black,below left] at (xone) {$x_1$};
  \node[blue!80!black,above right] at (yone) {$y_1$};
  \node[blue!80!black,right,fill=white,inner sep=1pt]
    at (0.605,0.438) {\scriptsize $t_+$};
  \node[blue!80!black,above,fill=white,inner sep=1pt]
    at (0.783,0.605) {\scriptsize $t_-$};

  % Negative-first path: (0.1,0.3) -> (0.7,0.9).
  % t_- = 1/2, t_+ = 5/6.
  \coordinate (xtwo) at (0.10,0.30);
  \coordinate (ytwo) at (0.70,0.90);
  \coordinate (pminustwo) at (0.40,0.60);
  \coordinate (pplustwo) at (0.60,0.80);
  \draw[red!70!yellow!85!black,very thick,-{Latex[length=2.6mm]}]
    (xtwo) -- (ytwo);
  \filldraw[red!70!yellow!85!black] (xtwo) circle (2.2pt);
  \filldraw[red!70!yellow!85!black] (ytwo) circle (2.2pt);

  \filldraw[draw=red!70!yellow!85!black,fill=white,very thick]

  \node[red!65!black,left,fill=white,inner sep=1pt]
    at (0.595,0.80) {\scriptsize $t_+$};

  \coordinate (ythree) at (0.80,0.80);

  \node[
    blue!50!red,

  \node[anchor=west,align=left] at (0.115,-0.19)

  \node[anchor=west,align=left] at (0.115,-0.25)

\caption{Affine phase geometry for three $\Nval\to\Bval$ support trajectories at threshold $c=0.6$. Positive-first and negative-first paths necessarily occupy $\Tval$ and $\Fval$, respectively, between their unique wall-crossing times. The simultaneous path is the exceptional case $t_+=t_-$ and passes through the wall intersection $(c,c)$. The figure visualizes the Lean-verified crossing-order and intermediate-phase theorems; it does not by itself assert that every interpolated support point is already realized by a full admissible model.}
\label{fig:fde-phase-geometry}
\end{figure}
